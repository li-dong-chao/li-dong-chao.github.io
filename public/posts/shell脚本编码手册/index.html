shell脚本打印带颜色日志

 1SETCOLOR_SUCCESS="echo -n \\033[1;32m"
 2SETCOLOR_FAILURE="echo -n \\033[1;31m"
 3SETCOLOR_WARNING="echo -n \\033[1;33m"
 4SETCOLOR_NORMAL="echo -n \\033[0;39m"
 5
 6LogWarnMsg()
 7{
 8    time=$(date "+%D %T")
 9    $SETCOLOR_WARNING
10    echo "[$time] : WARN    : $*"
11    $SETCOLOR_NORMAL
12}
13
14LogSucMsg()
15{
16    time=$(date "+%D %T")
17    $SETCOLOR_SUCCESS
18    echo "[$time] : SUCCESS : $*"
19    $SETCOLOR_NORMAL
20}
21
22LogErrorMsg()
23{
24    time=$(date "+%D %T")
25    $SETCOLOR_FAILURE
26    echo "[$time] : ERROR   : $*"
27    $SETCOLOR_NORMAL
28}
29
30LogSucMsg "success"
31LogWarnMsg "warning"
32LogErrorMsg "error"
