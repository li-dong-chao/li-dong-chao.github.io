C#与Python实现rsa加密解密

05 Aug 2023

最近在项目中遇到一个需求，需要在客户端将数据加密，然后传到服务端进行解密。客户端逻辑是用 C# 实现的，然后服务端逻辑是用 Python 实现的，加密方式要求用rsa非对称加密。

由于 C# 使用的私钥、公钥文件和 Python 生成的私钥、公钥文件不同，所以需要实现两者的转化。这里记录一下两者的转化过程。

C#实现rsa非对称加密

首先，可以使用 C# 创建私钥、公钥，并保存到 xml 格式的文件中（ C# 生成的rsa密钥是 xml 格式的）。代码如下：

 1using System;
 2using System.IO;
 3using System.Security.Cryptography;
 4using System.Text;
 5namespace TestConsoleApplication
 6{
 7    public class RSAHelper
 8    {
 9        // 创建rsa公钥、私钥
10        public static void GenerateRSAKey(string rsaFile, string rsaPubFile)
11        {
12            var rsa = new RSACryptoServiceProvider(2048);
13            // 创建私钥
14            string a = rsa.ToXmlString(true);
15            // 创建公钥
16            string b = rsa.ToXmlString(false);
17            // 保存公钥和私钥
18            try
19            {
20                StreamWriter sw1 = new StreamWriter(rsaFile);
21                sw1.WriteLine(a);
22                sw1.Close();
23                StreamWriter sw2 = new StreamWriter(rsaPubFile);
24                sw2.WriteLine(b);
25                sw2.Close();
26            }
27            catch (Exception e)
28            {
29                Console.WriteLine("Exception: " + e.Message);
30            }
31            finally
32            {
33                Console.WriteLine("Executing finally block.");
34            }
35        }
36    }
37    class Program
38    {
39        // 主程序
40        static void Main(string[] args)
41        {
42            // 创建用于加密的公钥私钥
43            string rsaFile = "rsa.xml";  // 保存私钥的文件
44            string rsaPubFile = "rsa_pub.xml";  // 保存公钥的文件
45            RSAHelper.GenerateRSAKey(rsaFile, rsaPubFile);
46        }
47    }
48}

运行上面的代码，可以得到两个文件 rsa.xml 和 rsa_pub.xml ，这两个文件分别存储了私钥和公钥信息。

下面继续完善一下前面代码中的 RSAHelper 类，使其能够进行rsa加密。

 37            var byteConverter = new UTF8Encoding();

公钥私钥的转化

由于 C# 生成的公钥私钥是 xml 格式的，与 python 使用的公钥私钥格式不同，所以需要对生成的公钥私钥进行转化。 转化的方式可以参考下面的步骤：

  1. 保存 XMLSec2PEM.java 文件到本地； 该文件的内容如下：
  1import java.io.*;
  2import java.math.BigInteger;
  3import java.security.*;
  4import java.security.spec.*;
  5import java.security.cert.*;
  6import java.security.interfaces.*;
  7
  8import javax.xml.parsers.DocumentBuilder;
  9import javax.xml.parsers.DocumentBuilderFactory;
 10import org.xml.sax.SAXException;
 11import org.xml.sax.SAXParseException;
 12import org.w3c.dom.*;
 13import org.w3c.dom.DOMException;
 14
 15/**
 16 * Converts a private or public RSA key from the XML Security format (used also
 17 * in .NET) to the RSA key format in PEM format. 
 18 *
 19 * The output key is written on standard output.
 20 *
 21 * In case of private key, the output format is the "traditional" format of the 
 22 * key, i.e. NOT the newer PKCS#8 format. The output keys are unencrypted. To
 23 * convert to PKCS#8 and/or encrypt the private key, use openssl with the
 24 * -topk8 option: 
 25 *    openssl pkcs8 -topk8 -in privkey_rsa.pem -out privkey.pem
 26 * 
 27 * Created by matous.borak@platanus.cz, 2008.
 28 * See http://www.platanus.cz/blog/xml_key_to_pem_en.html
 29 *
 30 * Heavily inspired by the PvkConvert utility made by Michel Gallant, see
 31 *   http://www.jensign.com/JavaScience/PvkConvert/. Thanks for it!
 32 *
 33 */
 34public class XMLSec2PEM {
 35
 36  private static final int PRIVATE_KEY = 1;
 37  private static final int PUBLIC_KEY = 2;
 38  private static final String[] PRIVATE_KEY_XML_NODES =  { "Modulus", "Exponent", "P", "Q", "DP", "DQ", "InverseQ", "D" };
 39  private static final String[] PUBLIC_KEY_XML_NODES =  { "Modulus", "Exponent" };
 40
 41  public static void main(String args[]) {
 42
 43    if (args.length != 1) {
 44      System.out.println("Usage:\n  java XMLSec2PEM <XMLSecurityRSAKeyValueFile.xml>");
 45      System.exit(0) ;
 46    }
 47
 48    try {
 49
 50      Document XMLSecKeyDoc = parseXMLFile(args[0]);
 51      System.out.print("Determining the key type: ");
 52      int keyType = getKeyType(XMLSecKeyDoc);
 53      if (keyType == PRIVATE_KEY || keyType == PUBLIC_KEY) {
 54        System.out.println("seems to be a " + ( keyType == PRIVATE_KEY ? "private" : "public" ) + " XML Security key");
 55      } else {
 56        System.exit(1);
 57      }
 58
 59      System.out.print("Checking the XML file structure: ");
 60      if (checkXMLRSAKey(keyType, XMLSecKeyDoc)) {
 61        System.out.println("OK");
 62      } else {
 63        System.exit(1);
 64      }
 65
 66      String pem = "";
 67      System.out.println("Outputting the resulting key:\n");
 68      if (keyType == PRIVATE_KEY) {
 69        pem = convertXMLRSAPrivateKeyToPEM(XMLSecKeyDoc);
 70        System.out.println("-----BEGIN PRIVATE KEY-----");
 71        System.out.println(pem);
 72        System.out.println("-----END PRIVATE KEY-----");
 73      } else {
 74        pem = convertXMLRSAPublicKeyToPEM(XMLSecKeyDoc);
 75        System.out.println("-----BEGIN PUBLIC KEY-----");
 76        System.out.println(pem);
 77        System.out.println("-----END PUBLIC KEY-----");
 78      }
 79
 80    }
 81
 82    catch (Exception e) {
 83      System.err.println(e);
 84    }
 85  }
 86
 87  private static int getKeyType(Document xmldoc) {
 88
 89    Node root = xmldoc.getFirstChild();
 90    if (!root.getNodeName().equals("RSAKeyValue")) {
 91      System.out.println("Expecting <RSAKeyValue> node, encountered <" + root.getNodeName() + ">");
 92      return 0;
 93    }
 94    NodeList children = root.getChildNodes();
 95    if (children.getLength() == PUBLIC_KEY_XML_NODES.length) {
 96      return PUBLIC_KEY;
 97    } 
 98    return PRIVATE_KEY;
 99  }
100
101  private static boolean checkXMLRSAKey(int keyType, Document xmldoc) {
102
103    Node root = xmldoc.getFirstChild();
104    NodeList children = root.getChildNodes();
105    String[] wantedNodes = {};
106    if (keyType == PRIVATE_KEY) {
107      wantedNodes = PRIVATE_KEY_XML_NODES;
108    } else {
109      wantedNodes = PUBLIC_KEY_XML_NODES;
110    }
111    for (int j = 0; j < wantedNodes.length; j++) {
112      String wantedNode = wantedNodes[j];
113      boolean found = false;
114      for (int i = 0; i < children.getLength(); i++) {
115        if (children.item(i).getNodeName().equals(wantedNode)) {
116          found = true;
117          break;
118        }
119      }
120      if (!found) {
121        System.out.println("Cannot find node <" + wantedNode + ">");
122        return false;
123      }
124    }
125    return true;
126  }
127
128  private static String convertXMLRSAPrivateKeyToPEM(Document xmldoc) {
129
130    Node root = xmldoc.getFirstChild();
131    NodeList children = root.getChildNodes();
132
133    BigInteger modulus = null, exponent = null, primeP = null, primeQ = null, 
134               primeExponentP = null, primeExponentQ = null, 
135               crtCoefficient = null, privateExponent = null;
136
137    for (int i = 0; i < children.getLength(); i++) {
138
139      Node node = children.item(i);
140      String textValue = node.getTextContent();
141      if (node.getNodeName().equals("Modulus")) {
142        modulus = new BigInteger(b64decode(textValue));
143      } else if (node.getNodeName().equals("Exponent")) {
144        exponent = new BigInteger(b64decode(textValue));
145      } else if (node.getNodeName().equals("P")) {
146        primeP = new BigInteger(b64decode(textValue));
147      } else if (node.getNodeName().equals("Q")) {
148        primeQ = new BigInteger(b64decode(textValue));
149      } else if (node.getNodeName().equals("DP")) {
150        primeExponentP = new BigInteger(b64decode(textValue));
151      } else if (node.getNodeName().equals("DQ")) {
152        primeExponentQ = new BigInteger(b64decode(textValue));
153      } else if (node.getNodeName().equals("InverseQ")) {
154        crtCoefficient = new BigInteger(b64decode(textValue));
155      } else if (node.getNodeName().equals("D")) {
156        privateExponent = new BigInteger(b64decode(textValue));
157      }
158    }
159
160    try {
161
162      RSAPrivateCrtKeySpec keySpec = new RSAPrivateCrtKeySpec (
163        modulus, exponent, privateExponent, primeP, primeQ, 
164        primeExponentP, primeExponentQ, crtCoefficient);
165
166      KeyFactory keyFactory = KeyFactory.getInstance("RSA");
167      PrivateKey key = keyFactory.generatePrivate(keySpec);
168      return b64encode(key.getEncoded());
169
170    } catch (Exception e) {
171      System.out.println(e);
172    }
173    return null;
174  }
175
176  private static String convertXMLRSAPublicKeyToPEM(Document xmldoc) {
177
178    Node root = xmldoc.getFirstChild();
179    NodeList children = root.getChildNodes();
180
181    BigInteger modulus = null, exponent = null;
182
183    for (int i = 0; i < children.getLength(); i++) {
184
185      Node node = children.item(i);
186      String textValue = node.getTextContent();
187      if (node.getNodeName().equals("Modulus")) {
188        modulus = new BigInteger(b64decode(textValue));
189      } else if (node.getNodeName().equals("Exponent")) {
190        exponent = new BigInteger(b64decode(textValue));
191      }
192    }
193
194    try {
195
196      RSAPublicKeySpec keySpec = new RSAPublicKeySpec(modulus, exponent);
197
198      KeyFactory keyFactory = KeyFactory.getInstance("RSA");
199      PublicKey key = keyFactory.generatePublic(keySpec);
200      return b64encode(key.getEncoded());
201
202    } catch (Exception e) {
203      System.out.println(e);
204    }
205    return null;
206
207  }
208
209  private static Document parseXMLFile(String filename) {
210    try {
211      DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
212      DocumentBuilder builder = factory.newDocumentBuilder();
213      Document document = builder.parse( new File(filename) );
214      return document;
215    } catch(Exception e) {
216      System.err.println(e); 
217      return null;
218    }  
219  }
220
221  private static final String b64encode(byte[] data) {
222    sun.misc.BASE64Encoder enc = new sun.misc.BASE64Encoder() ;
223    String b64str = enc.encodeBuffer(data).trim();
224    return b64str;
225  }
226
227  private static final byte[] b64decode(String data) {
228    try {
229      sun.misc.BASE64Decoder dec = new sun.misc.BASE64Decoder() ;
230      byte[] bytes = dec.decodeBuffer(data.trim());
231      return bytes;
232    } catch (IOException e) {
233      System.out.println("Exception caught when base64 decoding!" + e.toString());
234    }
235    return null;
236  }

  1. 编译java代码（注意：要求java版本为1.8，java版本过高可能会报错）
1javac XMLSec2PEM.java
  1. 转换 xml 到 pem
1java XMLSec2PEM your_key.xml
  1. 将上述输出的内容保存到文件 rsa_key.pem
  2. 使用 openssl 工具对 pem 文件进行转化，否则 python 无法直接使用
1openssl pkcs8 -topk8 -in rsa_key.pem -nocrypt -out rsa.key
  1. 根据 rsa.key 文件，可以进一步使用 openssl 工具提取出公钥
1openssl rsa -in rsa.key -pubout -out rsa_pub.key

执行完以上6步，便可以得到私钥 rsa.key 和公钥 rsa_pub.key 了。 python 使用上述两个文件进行加密解密即可。

使用Python进行rsa加密解密

 1from Crypto.PublicKey import RSA
 2from Crypto.Cipher import PKCS1_v1_5 as Cipher_pkcs1_v1_5
 3import base64
 4
 5# 加密
 6message = "你好，李银河"
 7rsakey = RSA.importKey(open("rsa_pub.key").read())  # 这里使用前面生成的公钥
 8cipher = Cipher_pkcs1_v1_5.new(rsakey)  # 创建用于执行pkcs1_v1_5加密或解密的密码
 9cipher_text = base64.b64encode(cipher.encrypt(message.encode('utf-8')))
10print(cipher_text.decode('utf-8'))
11
12# 解密
13cipher_text = cipher_text.decode('utf-8')
14encrypt_text = cipher_text.encode('utf-8')
15rsakey = RSA.importKey(open("rsa.key").read())  # 这里使用前面生成的私钥
16cipher = Cipher_pkcs1_v1_5.new(rsakey)  # 创建用于执行pkcs1_v1_5加密或解密的密码
17text = cipher.decrypt(base64.b64decode(encrypt_text), "解密失败")
18print(text.decode('utf-8'))
